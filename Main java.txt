// 1. CLASE ABSTRACTA PERSONA (Práctica 05 y 06 - Abstracción y Herencia)
// =========================================================================
// Declaramos la clase Persona como abstracta porque no queremos instanciar personas genéricas,
// sino únicamente tipos específicos como Usuarios o Bibliotecarios.
abstract class Persona {
    // Atributo protegido para almacenar el ID identificador de la persona
    protected int id;
    // Atributo protegido para el nombre completo de la persona
    protected String nombre;
    // Atributo protegido para el correo electrónico de contacto
    protected String correo;

    // Constructor por defecto que inicializa los atributos con valores neutros
    public Persona() {
        // Asignamos 0 como ID por defecto
        this.id = 0;
        // Asignamos una cadena genérica para el nombre
        this.nombre = "Sin nombre";
        // Asignamos un correo genérico
        this.correo = "sin@correo.com";
    }

    // Constructor con parámetros para inicializar una persona con datos específicos
    public Persona(int id, String nombre, String correo) {
        // Asignamos el valor del parámetro id al atributo de la clase
        this.id = id;
        // Asignamos el nombre recibido al atributo correspondiente
        this.nombre = nombre;
        // Asignamos el correo recibido al atributo correspondiente
        this.correo = correo;
    }

    // Método abstracto obligatorio que obliga a las subclases a definir cómo muestran su información
    public abstract void mostrarInformacion();

    // Método getter para obtener el ID de la persona
    public int getId() { return id; }
    // Método setter para actualizar el ID de la persona
    public void setId(int id) { this.id = id; }

    // Método getter para obtener el nombre de la persona
    public String getNombre() { return nombre; }
    // Método setter para actualizar el nombre de la persona
    public void setNombre(String nombre) { this.nombre = nombre; }

    // Método getter para obtener el correo de la persona
    public String getCorreo() { return correo; }
    // Método setter para actualizar el correo de la persona
    public void setCorreo(String correo) { this.correo = correo; }
}

// 2. CLASE USUARIO (Práctica 06 - Herencia)
// =========================================================================
// La clase Usuario hereda los atributos y métodos públicos/protegidos de Persona
class Usuario extends Persona {
    // Contador privado que lleva el registro de cuántos libros tiene prestados actualmente
    private int librosPrestados;
    // Cadena privada para almacenar el número telefónico
    private String telefonoCelular;
    // Bandera booleana para saber si el usuario está activo o suspendido en el sistema
    private boolean activo;

    // Constructor por defecto
    public Usuario() {
        // Llamamos al constructor por defecto de la clase madre (Persona)
        super();
        // Inicializamos el contador de libros prestados en 0
        this.librosPrestados = 0;
        // Asignamos un teléfono genérico por defecto
        this.telefonoCelular = "00-00-00-00-00";
        // Definimos el estado del usuario como activo por defecto
        this.activo = true;
    }

    // Primer constructor sobrecargado (recibe id, nombre, correo y estado activo)
    public Usuario(int id, String nombre, String correo, boolean activo) {
        // Invocamos al constructor de Persona pasándole id, nombre y correo
        super(id, nombre, correo);
        // Inicializamos el contador de libros prestados en 0
        this.librosPrestados = 0;
        // Establecemos el teléfono como no registrado
        this.telefonoCelular = "No registrado";
        // Asignamos el valor del estado activo recibido
        this.activo = activo;
    }

    // Segundo constructor sobrecargado (recibe id, nombre, correo, teléfono y estado activo)
    public Usuario(int id, String nombre, String correo, String telefonoCelular, boolean activo) {
        // Enviamos los datos generales a la clase padre mediante super
        super(id, nombre, correo);
        // Inicializamos el contador de libros prestados en 0
        this.librosPrestados = 0;
        // Asignamos el número celular recibido
        this.telefonoCelular = telefonoCelular;
        // Asignamos el estado de actividad del usuario
        this.activo = activo;
    }

    // Método para simular la solicitud de un préstamo por parte del usuario
    public void solicitarPrestamo() {
        // Verificamos si el usuario se encuentra con estatus activo
        if (activo) {
            // Incrementamos la cantidad de libros que tiene prestados
            librosPrestados++;
            // Imprimimos la confirmación en consola indicando la nueva cantidad de préstamos
            System.out.println("-> El usuario " + nombre + " ha solicitado un préstamo. Libros prestados actualmente: " + librosPrestados);
        } else {
            // Si el usuario no está activo, mostramos un mensaje de restricción
            System.out.println("-> El usuario " + nombre + " no está activo.");
        }
    }

    // Método para simular la devolución de un libro prestado
    public void devolverLibro() {
        // Comprobamos si el usuario tiene al menos un libro en su poder
        if (librosPrestados > 0) {
            // Decrementamos la cuenta de libros prestados
            librosPrestados--;
            // Mostramos la confirmación en consola indicando la cantidad restante
            System.out.println("-> El usuario " + nombre + " devolvió un libro. Libros pendientes: " + librosPrestados);
        } else {
            // Si el contador está en 0, indicamos que no tiene adeudos de libros
            System.out.println("-> El usuario " + nombre + " no tiene libros pendientes por devolver.");
        }
    }

    // Sobreescritura del método abstracto declarado en Persona
    @Override
    public void mostrarInformacion() {
        // Imprimimos todos los datos concatenados del usuario
        System.out.println("[USUARIO] ID: " + id + " | Nombre: " + nombre + " | Correo: " + correo +
                " | Teléfono: " + telefonoCelular + " | Activo: " + activo +
                " | Libros prestados: " + librosPrestados);
    }

    // Getter para consultar la cantidad de libros prestados
    public int getLibrosPrestados() { return librosPrestados; }
    // Getter para consultar el teléfono celular
    public String getTelefonoCelular() { return telefonoCelular; }
    // Setter para modificar el teléfono celular
    public void setTelefonoCelular(String telefonoCelular) { this.telefonoCelular = telefonoCelular; }
    // Getter para validar el estatus de la cuenta del usuario
    public boolean isActivo() { return activo; }
    // Setter para cambiar el estatus de la cuenta del usuario
    public void setActivo(boolean activo) { this.activo = activo; }
}

// 4. CLASE LIBRO (Práctica 04 y 05 - Encapsulamiento, Sobrecarga y Static)
// =========================================================================
// Clase concreta que modela un libro físico dentro del inventario
class Libro {
    // Atributo privado para el código estándar internacional ISBN
    private String isbn;
    // Atributo privado para el título de la obra
    private String titulo;
    // Atributo privado para el autor o autores de la obra
    private String autor;
    // Atributo privado para el año de edición/publicación
    private int anioPublicacion;
    // Atributo privado para controlar si el libro está libre o ya prestado
    private boolean disponible;

    // Variable estática compartida por todos los objetos Libro para contar cuántos se han creado
    private static int totalLibros = 0;

    // Constructor sin parámetros
    public Libro() {
        // Asignamos un valor nulo ficticio para el ISBN
        this.isbn = "0000000000";
        // Asignamos el título genérico
        this.titulo = "Sin título";
        // Asignamos un autor por defecto
        this.autor = "Anónimo";
        // Asignamos 0 al año
        this.anioPublicacion = 0;
        // Por defecto, el libro recién creado estará disponible
        this.disponible = true;
        // Incrementamos el contador global estático de libros instanciados
        totalLibros++;
    }

    // Primer constructor sobrecargado (recibe solo título y autor)
    public Libro(String titulo, String autor) {
        // Asignamos un ISBN por defecto
        this.isbn = "0000000000";
        // Asignamos el título recibido
        this.titulo = titulo;
        // Asignamos el autor recibido
        this.autor = autor;
        // Asignamos el año actual por defecto
        this.anioPublicacion = 2024;
        // Marcamos como disponible
        this.disponible = true;
        // Incrementamos la variable estática contadora
        totalLibros++;
    }

    // Segundo constructor sobrecargado (recibe ISBN, título, autor y año de publicación)
    public Libro(String isbn, String titulo, String autor, int anioPublicacion) {
        // Asignamos el código ISBN recibido
        this.isbn = isbn;
        // Asignamos el título recibido
        this.titulo = titulo;
        // Asignamos el autor del libro
        this.autor = autor;
        // Asignamos el año de publicación
        this.anioPublicacion = anioPublicacion;
        // Establecemos disponibilidad en verdadero
        this.disponible = true;
        // Sumamos uno al acumulador estático de libros
        totalLibros++;
    }

    // Método para verificar la disponibilidad del ejemplar
    public boolean estaDisponible() { return disponible; }

    // Método para marcar que el libro fue prestado
    public void prestar() {
        // Revisamos si el libro está libre para prestarse
        if (disponible) {
            // Cambiamos el estado a no disponible
            disponible = false;
            // Desplegamos el mensaje de préstamo exitoso
            System.out.println("-> El libro '" + titulo + "' ha sido prestado exitosamente.");
        } else {
            // Avisamos que el libro ya se encuentra ocupado
            System.out.println("-> El libro '" + titulo + "' ya se encuentra prestado actualmente.");
        }
    }

    // Método para registrar la devolución del libro a la estantería
    public void devolver() {
        // Restauramos el estatus a disponible
        disponible = true;
        // Mostramos el mensaje en consola
        System.out.println("-> El libro '" + titulo + "' ha sido marcado como disponible.");
    }

    // Getter para ISBN
    public String getIsbn() { return isbn; }
    // Setter para ISBN
    public void setIsbn(String isbn) { this.isbn = isbn; }
    // Getter para el Título
    public String getTitulo() { return titulo; }
    // Setter para el Título
    public void setTitulo(String titulo) { this.titulo = titulo; }
    // Getter para el Autor
    public String getAutor() { return autor; }
    // Setter para el Autor
    public void setAutor(String autor) { this.autor = autor; }
    // Getter para el Año de publicación
    public int getAnioPublicacion() { return anioPublicacion; }
    // Setter para el Año de publicación
    public void setAnioPublicacion(int anioPublicacion) { this.anioPublicacion = anioPublicacion; }
    // Setter directo para la disponibilidad
    public void setDisponible(boolean disponible) { this.disponible = disponible; }

    // Método estático de clase para consultar el contador global de libros creados
    public static int getTotalLibros() { return totalLibros; }

    // Método auxiliar para formatear e imprimir la información completa del objeto Libro
    public void mostrarInformacion() {
        // Imprimimos la ficha técnica con todos sus atributos encapsulados
        System.out.println("ISBN: " + isbn + " | Título: " + titulo + " | Autor: " + autor +
                " | Año: " + anioPublicacion + " | Disponible: " + disponible);
    }
}

// 5. CLASE PRESTAMO (Práctica 07 - Comunicación entre Objetos)
// =========================================================================
// La clase Prestamo modela la relación e interacción entre un objeto Usuario y un objeto Libro
class Prestamo {
    // Objeto contenido de tipo Usuario que realiza la solicitud
    private Usuario usuario;
    // Objeto contenido de tipo Libro que es solicitado
    private Libro libro;
    // Fecha en la que se registra el inicio del préstamo
    private String fechaPrestamo;
    // Fecha pactada para la devolución del libro
    private String fechaDevolucion;
    // Bandera para indicar si el registro de préstamo se encuentra actualmente en curso
    private boolean activo;

    // Constructor que recibe las instancias del usuario y del libro
    public Prestamo(Usuario usuario, Libro libro) {
        // Asignamos el objeto Usuario recibido
        this.usuario = usuario;
        // Asignamos el objeto Libro recibido
        this.libro = libro;
        // Asignamos una fecha genérica para la simulación
        this.fechaPrestamo = "Hoy";
        // Asignamos la fecha de entrega por defecto
        this.fechaDevolucion = "En 7 días";
        // Inicializamos la bandera activo en false hasta ejecutar la transacción
        this.activo = false;
    }

    // Método para iniciar el préstamo y provocar la comunicación directa entre objetos
    public void iniciarPrestamo() {
        // Comprobamos si el libro está disponible y el usuario está activo
        if (libro.estaDisponible() && usuario.isActivo()) {
            // El objeto Prestamo le indica a la instancia Libro que ejecute su método prestar()
            libro.prestar();
            // El objeto Prestamo le indica al Usuario que incremente su lista mediante solicitarPrestamo()
            usuario.solicitarPrestamo();
            // Cambiamos el estatus del registro a activo
            this.activo = true;
            // Confirmamos la transacción completa
            System.out.println("-> Transacción exitosa: Préstamo activado.");
        } else {
            // Avisamos si no fue posible concretar el préstamo
            System.out.println("-> No se pudo iniciar el préstamo. Verifique el estatus del libro o usuario.");
        }
    }

    // Método para dar por concluido el préstamo
    public void finalizarPrestamo() {
        // Validamos que la transacción siga activa
        if (activo) {
            // Ordenamos al libro que marque su estatus como disponible
            libro.devolver();
            // Ordenamos al usuario que reduzca su contador de préstamos
            usuario.devolverLibro();
            // Desactivamos el registro del préstamo
            this.activo = false;
            // Confirmamos la finalización
            System.out.println("-> Transacción exitosa: Préstamo finalizado.");
        } else {
            // Avisamos si el préstamo ya había sido cerrado antes
            System.out.println("-> Este préstamo ya no se encuentra activo.");
        }
    }

    // Getter para consultar si el préstamo sigue activo
    public boolean estaActivo() { return activo; }
    // Getter para obtener el objeto Usuario
    public Usuario getUsuario() { return usuario; }
    // Getter para obtener el objeto Libro
    public Libro getLibro() { return libro; }
    // Getter para obtener la fecha de préstamo
    public String getFechaPrestamo() { return fechaPrestamo; }
    // Getter para obtener la fecha de devolución
    public String getFechaDevolucion() { return fechaDevolucion; }
}

// 6. CLASE BIBLIOTECA (Práctica 07 - Composición y Uso de ArrayList)
// =========================================================================
// Clase central que administra el inventario, catálogo y registros mediante ArrayList
class Biblioteca {
    // Lista dinámica para almacenar objetos de tipo Libro
    private ArrayList<Libro> libros;
    // Lista dinámica para almacenar objetos de tipo Usuario
    private ArrayList<Usuario> usuarios;
    // Lista dinámica para almacenar el personal de Bibliotecarios
    private ArrayList<Bibliotecario> empleados;
    // Lista dinámica para guardar el historial de objetos Prestamo
    private ArrayList<Prestamo> prestamos;

    // Constructor que se encarga de instanciar todas las listas vacías
    public Biblioteca() {
        // Inicializamos el ArrayList de libros
        this.libros = new ArrayList<>();
        // Inicializamos el ArrayList de usuarios
        this.usuarios = new ArrayList<>();
        // Inicializamos el ArrayList de empleados
        this.empleados = new ArrayList<>();
        // Inicializamos el ArrayList de préstamos
        this.prestamos = new ArrayList<>();
    }

    // Método para añadir un libro al catálogo de la biblioteca
    public void agregarLibro(Libro libro) {
        // Agregamos el libro recibido a la lista
        libros.add(libro);
        // Notificamos la adición exitosa
        System.out.println("-> Libro '" + libro.getTitulo() + "' agregado a la biblioteca.");
    }

    // Método para dar de baja un libro mediante su código ISBN
    public void eliminarLibro(String isbn) {
        // Buscamos la referencia del libro en el catálogo
        Libro l = buscarLibro(isbn);
        // Si el libro fue encontrado lo removemos de la lista
        if (l != null) {
            // Removemos la instancia
            libros.remove(l);
            // Confirmamos la eliminación
            System.out.println("-> Libro con ISBN " + isbn + " eliminado correctamente.");
        } else {
            // Notificamos si no se encontró en la lista
            System.out.println("-> No se encontró ningún libro con ese ISBN.");
        }
    }

    // Método de búsqueda para localizar un libro por su ISBN
    public Libro buscarLibro(String isbn) {
        // Recorremos el ArrayList de libros mediante un ciclo for-each
        for (Libro l : libros) {
            // Comparamos el ISBN ignorando diferencias entre mayúsculas y minúsculas
            if (l.getIsbn().equalsIgnoreCase(isbn)) {
                // Devolvemos el objeto hallado
                return l;
            }
        }
        // Retornamos null si el bucle termina sin coincidencias
        return null;
    }

    // Sobrecarga del método buscarLibro para localizar una obra usando Título y Autor
    public Libro buscarLibro(String titulo, String autor) {
        // Recorremos la lista de libros
        for (Libro l : libros) {
            // Validamos coincidencia simultánea de título y autor
            if (l.getTitulo().equalsIgnoreCase(titulo) && l.getAutor().equalsIgnoreCase(autor)) {
                // Devolvemos la referencia encontrada
                return l;
            }
        }
        // Retornamos null si no coincide ningún elemento
        return null;
    }

    // Método para imprimir en pantalla el catálogo completo de la biblioteca
    public void mostrarCatalogo() {
        System.out.println("\n--- CATÁLOGO COMPLETO DE LIBROS ---");
        // Revisamos si la lista está vacía
        if (libros.isEmpty()) {
            System.out.println("El catálogo de la biblioteca está vacío.");
        } else {
            // Recorremos la lista llamando al método mostrarInformacion() de cada objeto
            for (Libro l : libros) {
                l.mostrarInformacion();
            }
        }
    }

    // Método para dar de alta a un usuario dentro del sistema
    public void registrarUsuario(Usuario usuario) {
        // Agregamos la instancia al ArrayList de usuarios
        usuarios.add(usuario);
        // Confirmamos el alta
        System.out.println("-> Usuario '" + usuario.getNombre() + "' registrado correctamente.");
    }

    // Método para buscar un usuario a través de su ID
    public Usuario buscarUsuario(int id) {
        // Recorremos el ArrayList de usuarios registrados
        for (Usuario u : usuarios) {
            // Si coincide el ID devolvemos el objeto Usuario
            if (u.getId() == id) {
                return u;
            }
        }
        // Retornamos null si no existe
        return null;
    }

    // Método para imprimir todos los usuarios registrados
    public void mostrarUsuarios() {
        System.out.println("\n--- LISTA DE USUARIOS REGISTRADOS ---");
        // Revisamos si la lista no contiene registros
        if (usuarios.isEmpty()) {
            System.out.println("No hay usuarios registrados en el sistema.");
        } else {
            // Desplegamos la información de cada usuario
            for (Usuario u : usuarios) {
                u.mostrarInformacion();
            }
        }
    }

    // Método para registrar a un bibliotecario dentro de la lista de personal
    public void registrarEmpleado(Bibliotecario empleado) {
        // Agregamos al ArrayList de empleados
        empleados.add(empleado);
        // Confirmamos
        System.out.println("-> Bibliotecario '" + empleado.getNombre() + "' registrado correctamente.");
    }

    // Método para tramitar la transacción de un préstamo
    public void realizarPrestamo(Usuario usuario, Libro libro) {
        // Validamos que se reciban objetos válidos y no referencias nulas
        if (usuario != null && libro != null) {
            // Creamos un nuevo objeto de la clase Prestamo
            Prestamo p = new Prestamo(usuario, libro);
            // Iniciamos el proceso del préstamo
            p.iniciarPrestamo();
            // Si la operación concluyó exitosamente guardamos el préstamo en la lista
            if (p.estaActivo()) {
                prestamos.add(p);
            }
        } else {
            // Avisamos que no se ubicó a la persona o el libro
            System.out.println("-> Error al realizar préstamo: Usuario o libro no encontrado.");
        }
    }

    // Método para procesar la entrega/devolución de un libro en préstamo
    public void devolverLibro(Prestamo prestamo) {
        // Verificamos que el registro de préstamo sea válido y esté activo
        if (prestamo != null && prestamo.estaActivo()) {
            // Invocamos el cierre de la transacción en la clase Prestamo
            prestamo.finalizarPrestamo();
        } else {
            // Notificamos si el préstamo no es válido
            System.out.println("-> El préstamo especificado no es válido o ya fue devuelto.");
        }
    }

    // Método para desplegar todos los préstamos almacenados en la lista
    public void mostrarPrestamos() {
        System.out.println("\n--- LISTA DE PRÉSTAMOS REGISTRADOS ---");
        // Validamos si la lista no tiene registros
        if (prestamos.isEmpty()) {
            System.out.println("No hay préstamos activos o registrados.");
        } else {
            // Recorremos con ciclo for clásico para enumerar los préstamos
            for (int i = 0; i < prestamos.size(); i++) {
                // Obtenemos el elemento en el índice actual
                Prestamo p = prestamos.get(i);
                // Imprimimos el estado sintético del préstamo
                System.out.println("[" + (i + 1) + "] Usuario: " + p.getUsuario().getNombre() +
                        " | Libro: " + p.getLibro().getTitulo() +
                        " | Activo: " + p.estaActivo());
            }
        }
    }

    // Getter para retornar la lista completa de préstamos
    public ArrayList<Prestamo> getPrestamos() { return prestamos; }
}

// 7. CLASE PRINCIPAL (Punto de Entrada y Menú Consola Interactivo)
// =========================================================================
// Clase ejecutable que debe coincidir con el archivo Main.java
public class Main {
    // Método principal ejecutable main
    public static void main(String[] args) {
        // Inicializamos la lectura de datos por teclado mediante la clase Scanner
        Scanner sc = new Scanner(System.in);
        // Instanciamos el sistema principal mediante la clase Biblioteca
        Biblioteca miBiblioteca = new Biblioteca();

        // Variable para controlar las opciones seleccionadas en el menú principal
        int opcion = 0;

        // Bucle interactivo principal do-while
        do {
            // Desplegamos el menú con las funciones de la Práctica 1 a la 7
            System.out.println("\n==============================================");
            System.out.println("   SISTEMA DE GESTIÓN DE BIBLIOTECA (P01-P07) ");
            System.out.println("==============================================");
            System.out.println("1. Registrar un Libro en la Biblioteca");
            System.out.println("2. Registrar un Usuario en la Biblioteca");
            System.out.println("3. Registrar un Bibliotecario en el Sistema");
            System.out.println("4. Mostrar Catálogo de Libros");
            System.out.println("5. Mostrar Usuarios Registrados");
            System.out.println("6. Realizar Préstamo de Libro");
            System.out.println("7. Devolver Libro Prestado");
            System.out.println("8. Mostrar Lista de Préstamos");
            System.out.println("9. Buscar Libro (Demostración de Sobrecarga)");
            System.out.println("10. Ejecutar Casos de Prueba (Puntos 11.1 a 11.4)");
            System.out.println("11. Ver Total de Libros Creados (Atributo Static)");
            System.out.println("12. Salir");
            System.out.print("Selecciona una opción: ");

            // Validamos que el usuario introduzca un número entero
            if (sc.hasNextInt()) {
                // Leemos el valor seleccionado
                opcion = sc.nextInt();
                // Limpiamos la línea sobrante del buffer del Scanner
                sc.nextLine();
            } else {
                // Mensaje de advertencia si se introduce texto u otro tipo no entero
                System.out.println("Por favor ingresa un número válido.");
                // Limpiamos el buffer
                sc.nextLine();
                // Reiniciamos el ciclo
                continue;
            }

            // Selección de acciones según la opción elegida mediante switch
            switch (opcion) {
                // Opción 1: Alta de Libro interactiva
                case 1:
                    System.out.println("\n--- REGISTRAR LIBRO ---");
                    System.out.print("Ingresa el ISBN: ");
                    String isbn = sc.nextLine();
                    System.out.print("Ingresa el Título: ");
                    String titulo = sc.nextLine();
                    System.out.print("Ingresa el Autor: ");
                    String autor = sc.nextLine();
                    System.out.print("Ingresa el Año de Publicación: ");
                    int anio = sc.nextInt();
                    sc.nextLine();

                    // Instanciamos y agregamos el libro al catálogo de la biblioteca
                    Libro nuevoLibro = new Libro(isbn, titulo, autor, anio);
                    miBiblioteca.agregarLibro(nuevoLibro);
                    break;

                // Opción 2: Alta de Usuario interactiva
                case 2:
                    System.out.println("\n--- REGISTRAR USUARIO ---");
                    System.out.print("Ingresa el ID del usuario: ");
                    int idU = sc.nextInt();
                    sc.nextLine();
                    System.out.print("Ingresa el Nombre: ");
                    String nomU = sc.nextLine();
                    System.out.print("Ingresa el Correo: ");
                    String correoU = sc.nextLine();
                    System.out.print("Ingresa el Teléfono Celular: ");
                    String telU = sc.nextLine();

                    // Instanciamos el nuevo usuario y lo registramos en la biblioteca
                    Usuario nuevoUsuario = new Usuario(idU, nomU, correoU, telU, true);
                    miBiblioteca.registrarUsuario(nuevoUsuario);
                    break;

                // Opción 3: Alta de Bibliotecario
                case 3:
                    System.out.println("\n--- REGISTRAR BIBLIOTECARIO ---");
                    System.out.print("Ingresa el ID del bibliotecario: ");
                    int idB = sc.nextInt();
                    sc.nextLine();
                    System.out.print("Ingresa el Nombre: ");
                    String nomB = sc.nextLine();
                    System.out.print("Ingresa el Correo: ");
                    String correoB = sc.nextLine();
                    System.out.print("Ingresa el Turno (Matutino/Vespertino): ");
                    String turnoB = sc.nextLine();
                    System.out.print("Ingresa el Salario: ");
                    double salarioB = sc.nextDouble();
                    sc.nextLine();

                    // Instanciamos y registramos el bibliotecario en la lista de empleados
                    Bibliotecario nuevoEmpleado = new Bibliotecario(idB, nomB, correoB, turnoB, salarioB);
                    miBiblioteca.registrarEmpleado(nuevoEmpleado);
                    break;

                // Opción 4: Imprimir catálogo
                case 4:
                    miBiblioteca.mostrarCatalogo();
                    break;

                // Opción 5: Imprimir usuarios
                case 5:
                    miBiblioteca.mostrarUsuarios();
                    break;

                // Opción 6: Tramitar un préstamo entre un Usuario y un Libro registrados
                case 6:
                    System.out.println("\n--- TRAMITAR PRÉSTAMO ---");
                    System.out.print("Ingresa el ID del usuario: ");
                    int idBusq = sc.nextInt();
                    sc.nextLine();
                    System.out.print("Ingresa el ISBN del libro: ");
                    String isbnBusq = sc.nextLine();

                    // Buscamos los objetos en las listas de la biblioteca
                    Usuario uEnc = miBiblioteca.buscarUsuario(idBusq);
                    Libro lEnc = miBiblioteca.buscarLibro(isbnBusq);

                    // Solicitamos a la biblioteca procesar el préstamo
                    miBiblioteca.realizarPrestamo(uEnc, lEnc);
                    break;

                // Opción 7: Devolver un libro
                case 7:
                    System.out.println("\n--- DEVOLVER LIBRO PRESTADO ---");
                    // Desplegamos la lista de préstamos activos
                    miBiblioteca.mostrarPrestamos();
                    // Validamos si la lista contiene préstamos
                    if (!miBiblioteca.getPrestamos().isEmpty()) {
                        System.out.print("Ingresa el número de préstamo a finalizar: ");
                        int numP = sc.nextInt();
                        sc.nextLine();
                        // Ajustamos a índice 0 e invocamos la devolución
                        if (numP > 0 && numP <= miBiblioteca.getPrestamos().size()) {
                            Prestamo pDev = miBiblioteca.getPrestamos().get(numP - 1);
                            miBiblioteca.devolverLibro(pDev);
                        } else {
                            System.out.println("Número de préstamo no válido.");
                        }
                    }
                    break;

                // Opción 8: Mostrar la lista de préstamos
                case 8:
                    miBiblioteca.mostrarPrestamos();
                    break;

                // Opción 9: Demostración de búsqueda sobrecargada (Por ISBN o por Título y Autor)
                case 9:
                    System.out.println("\n--- BÚSQUEDA SOBRECARGADA DE LIBROS ---");
                    System.out.println("1. Buscar por ISBN");
                    System.out.println("2. Buscar por Título y Autor");
                    System.out.print("Selecciona el tipo de búsqueda: ");
                    int opB = sc.nextInt();
                    sc.nextLine();

                    if (opB == 1) {
                        System.out.print("Ingresa el ISBN: ");
                        String bIsbn = sc.nextLine();
                        // Llamada al método buscarLibro(String)
                        Libro res = miBiblioteca.buscarLibro(bIsbn);
                        if (res != null) res.mostrarInformacion();
                        else System.out.println("Libro no encontrado.");
                    } else if (opB == 2) {
                        System.out.print("Ingresa el Título: ");
                        String bTit = sc.nextLine();
                        System.out.print("Ingresa el Autor: ");
                        String bAut = sc.nextLine();
                        // Llamada al método sobrecargado buscarLibro(String, String)
                        Libro res = miBiblioteca.buscarLibro(bTit, bAut);
                        if (res != null) res.mostrarInformacion();
                        else System.out.println("Libro no encontrado.");
                    } else {
                        System.out.println("Opción no válida.");
                    }
                    break;

                // Opción 10: Ejecutar los casos de prueba requeridos en el documento (Secciones 11.1 a 11.4)
                case 10:
                    System.out.println("\n==============================================");
                    System.out.println("  EJECUTANDO CASOS DE PRUEBA DE LA PRÁCTICA   ");
                    System.out.println("==============================================");

                    // 11.1 Caso 1: Registro de 3 Libros
                    System.out.println("\n>>> Caso 1: Registro de Libros");
                    Libro lib1 = new Libro("101", "Java Básico", "Herbert Schildt", 2018);
                    Libro lib2 = new Libro("102", "Clean Code", "Robert C. Martin", 2008);
                    Libro lib3 = new Libro("103", "Patrones de Diseño", "Erich Gamma", 1994);
                    miBiblioteca.agregarLibro(lib1);
                    miBiblioteca.agregarLibro(lib2);
                    miBiblioteca.agregarLibro(lib3);

                    // 11.2 Caso 2: Registro de 2 Usuarios
                    System.out.println("\n>>> Caso 2: Registro de Usuarios");
                    Usuario us1 = new Usuario(1, "Juan Pérez", "juan@mail.com", true);
                    Usuario us2 = new Usuario(2, "María Gómez", "maria@mail.com", true);
                    miBiblioteca.registrarUsuario(us1);
                    miBiblioteca.registrarUsuario(us2);

                    // 11.3 Caso 3: Solicitud de Préstamo (Juan solicita 'Clean Code')
                    System.out.println("\n>>> Caso 3: Solicitud de Préstamo (Juan solicita Clean Code)");
                    System.out.println("Estado inicial de disponibildad de 'Clean Code': " + lib2.estaDisponible());
                    miBiblioteca.realizarPrestamo(us1, lib2);
                    System.out.println("Estado final de disponibildad de 'Clean Code': " + lib2.estaDisponible());

                    // 11.4 Caso 4: Devolución de Libro (Juan devuelve 'Clean Code')
                    System.out.println("\n>>> Caso 4: Devolución de Libro");
                    if (!miBiblioteca.getPrestamos().isEmpty()) {
                        // Obtenemos el último préstamo registrado
                        Prestamo pPrueba = miBiblioteca.getPrestamos().get(miBiblioteca.getPrestamos().size() - 1);
                        miBiblioteca.devolverLibro(pPrueba);
                    }
                    System.out.println("Estado final tras devolución de 'Clean Code': " + lib2.estaDisponible());
                    break;

                // Opción 11: Consultar el total estático
                case 11:
                    System.out.println("\nTotal de objetos 'Libro' instanciados (Atributo Static): " + Libro.getTotalLibros());
                    break;

                // Opción 12: Salida del programa
                case 12:
                    System.out.println("Saliendo del programa de gestión...");
                    break;

                // Opción por defecto para entradas fuera del rango 1-12
                default:
                    System.out.println("Opción no válida. Intenta de nuevo.");
            }
        // Condición de paro del bucle do-while
        } while (opcion != 12);

        // Cerramos la lectura del Scanner
        sc.close();
    }
}
